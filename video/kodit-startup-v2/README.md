## 다시 만드는 순서 (작업 폴더 — `../kodit-startup`의 cap · company · prod · fonts를 같이 씀)
```
venv/bin/python asr2.py ; venv/bin/python assemble.py → ffmpeg atempo 1.12 → highpass → loudnorm 2-pass → voice.wav
venv/bin/python asr.py ; python3 subs.py                  # OFFSET 0.8
python3 build.py h && python3 build.py v                  # comp/comp_*.html + out/sfx_*.json
node sublines.mjs ; node frames.mjs h fr <시각들> ; python3 sheet.py fr h a
venv/bin/python music.py ; ffmpeg highpass 70 + loudnorm 2-pass -24.5 LUFS → music.wav
bash render_all.sh h && bash render_all.sh v              # 60fps 4구간 → 섞기(mix.py: 목소리 + 음악 덕킹 + 효과음) → -14 LUFS → final_*.mp4
```
