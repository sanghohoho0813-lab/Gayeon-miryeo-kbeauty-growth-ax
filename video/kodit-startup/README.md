## 만드는 순서 (작업 폴더 = 이 폴더를 복사한 곳)
```
# 자료
cp <사용자 녹음>.wav src.wav
ffmpeg … loudnorm 2-pass → voice.wav                       # README 위 음량 기준
bash extract_company.sh .                                   # company/*.png (V7 영상에서)
cp ../../public/images/products/*.webp prod/                # 제품 이미지
cp -r ../../node_modules/pretendard/dist/web/variable/woff2/PretendardVariable.woff2 fonts/   # + fonts/pretendard.css
npm run build && npx next start -p 3100 ; node capture.mjs cap   # 앱 화면 (Demo 빌드, @2x) — PC는 고정 요소 그대로 긴 캡처
# 시각
venv/bin/python asr.py        # asr.json (단어 시각)
python3 subs.py               # subs.json/.srt + chars.json (대본 글자 → 음성 시각)
# 장면 → 렌더
python3 build.py h && python3 build.py v                    # comp/comp_h.html · comp_v.html
node sublines.mjs                                           # 자막 2줄 초과 검사
node frames.mjs h fr 6.3,13.7,… && python3 sheet.py fr h a  # 장면 끝 프레임 접촉 시트
bash render_all.sh h && bash render_all.sh v                # 4구간 병렬 → 이어 붙이기 → 음성(0.6초 여백 · 끝 1초 페이드 · +0.5dB 리미터)
```
